Types & format specifiers

int age = 25;         // %d
double price = 9.99;   // %f
char grade = 'A';     // %c
char name[40];        // %s

printf("%.2f\n", price);

Reading input

int number;
if (scanf("%d", &number) == 1) {
    printf("%d\n", number);
}
// & gives scanf the variable’s address.

Loops & decisions

for (int i = 0; i < 5; i++) {
    printf("%d\n", i);
}
if (age >= 18) { puts("Adult"); }
// == compares. = assigns.
// Arrays start at index 0.

Functions & pointers
