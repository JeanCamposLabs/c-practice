Your C pocket reference

The shape of a program

#include <stdio.h>

int main(void) {
    printf("Hello!\n");
    return 0;
}

Types & format specifiers

int age = 25;         // %d
double price = 9.99;   // %f
char grade = 'A';     // %c
char name[40];        // %s

printf("%.2f\n", price);

Reading input

int number;
if (scanf("%d", &number) == 1) {
    printf("%d\n", number);
}
// & gives scanf the variable’s address.
// Add input in the stdin panel or terminal.

Loops & decisions

for (int i = 0; i < 5; i++) {
    printf("%d\n", i);
}
if (age >= 18) { puts("Adult"); }
// == compares. = assigns.
// Arrays start at index 0.

Functions & pointers

int square(int n) { return n * n; }

int value = 7;
int *p = &value;
printf("%d\n", *p);  // 7
// Initialize pointers before using them.
